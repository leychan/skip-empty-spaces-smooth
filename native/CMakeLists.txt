cmake_minimum_required(VERSION 3.25)
project(SkipEmptySpacesNative LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(KWin 6.7 REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Qml DBus Widgets)
if(NOT DEFINED SPACES_BUILD)
    message(FATAL_ERROR "SPACES_BUILD must identify the source build")
endif()
string(SUBSTRING "${SPACES_BUILD}" 0 12 SPACES_SHORT)
set(SPACES_ID "skip-empty-spaces-native-${SPACES_SHORT}")
configure_file(metadata.json.in metadata.json @ONLY)
add_library(spacesrenderer MODULE renderer.cpp)
target_include_directories(spacesrenderer PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(spacesrenderer PRIVATE KWin::kwin Qt6::DBus)
target_compile_definitions(spacesrenderer PRIVATE SPACES_BUILD="${SPACES_BUILD}")
set_target_properties(spacesrenderer PROPERTIES PREFIX "" OUTPUT_NAME "${SPACES_ID}"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins/kwin/effects/plugins")
add_library(spacesbootstrap MODULE bootstrap.cpp)
target_link_libraries(spacesbootstrap PRIVATE KWin::kwin Qt6::Qml)
target_compile_definitions(spacesbootstrap PRIVATE SPACES_ID="${SPACES_ID}")
set_target_properties(spacesbootstrap PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml")
enable_testing()
add_executable(spaces_core_test core_test.cpp)
add_test(NAME spaces_core COMMAND spaces_core_test)
